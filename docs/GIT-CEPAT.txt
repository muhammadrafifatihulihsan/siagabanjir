# GIT BASH - salin per bagian. Berhenti jika ada error.
# Contoh Melia; Hasan/Rafi: ganti branch-melia dengan branch-hasan/branch-rafi.
# Living document: rules/documentation.md. Panduan lengkap: docs/GITHUB-WORKFLOW.md.

# 1. PERTAMA KALI: CLONE -> NGODING -> PUSH -> PR
git clone --branch branch-melia https://github.com/muhammadrafifatihulihsan/siagabanjir.git
cd siagabanjir
git branch -m main

# Ngoding. Setelah selesai, ganti path file dan pesan commit di bawah.
git add -- "path/file-yang-diubah"
git commit -m "feat: jelaskan perubahan"
git push origin main:branch-melia

# GitHub -> Pull requests -> New pull request
# base: development | compare: branch-melia -> Create pull request.

# 2. BERIKUTNYA: UPDATE MAIN -> NGODING -> PUSH -> PR
# Setelah Rafi merge PR ke development DAN promosinya ke main; folder kerja bersih.
git fetch origin
git merge --ff-only origin/main

# Ngoding lagi. Setelah selesai:
git add -- "path/file-yang-diubah"
git commit -m "feat: jelaskan perubahan berikutnya"
git push origin main:branch-melia

# Buat PR lagi: base development | compare branch-melia. Ulangi bagian 2.
